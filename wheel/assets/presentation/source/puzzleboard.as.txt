function handleShowPanels()

   _loc1_.status = mShowTarget;

         {
            _loc1_.status = "lLetter";
            _loc1_.gotoAndPlay("lLetter");

         {

         _loc1_.gotoAndPlay(mShowTarget);

         _loc1_.gotoAndPlay(mShowTarget);

function handleShowPanelsComplete()
{
   stopInterval();
   switch(mShowTarget)
   {
      case "lWhite":
         if(soeIsGame())
         {
         }
         fscommand("postShowWhitePanelsComplete",mFoundPanelsArray.length);
         break;
      case "lBlue":
         if(soeIsGame())
         {
         }
         doShowFoundLetters();
         break;
      case "lLetter":
         if(!soeIsGame())
         {
            soeTRACE("nbFoundLetters:" + nbFoundLetters);
         }
         if(!doCheckSolveForFoundLetters())
         {
            if(!doCheckVowels())
            {
               fscommand("postNoMoreVowels");
            }
            else if(!doCheckConsonants())
            {
               fscommand("postNoMoreConsonants");
            }
         }
         fscommand("postFindLettersComplete",nbFoundLetters);
      default:
         return;
   }
}

function onInterval()
{
   if(!s_bReady)
   {
      if(soeIsGame() && s_bSyncWithGame)
      {
         fscommand("postIsPuzzleboardRevealReady",mCurrentShowIndex);
      }
      else
      {
         s_bReady = true;
      }
   }
   if(s_bReveal && s_bReady)
   {
      s_bReady = false;
      s_bReveal = false;
      handleShowPanels();
      if(mCurrentShowIndex < mFoundPanelsArray.length)
      {
         startRevealTimer();
      }
      else
      {
         handleShowPanelsComplete();
      }
   }
}

function stopListeners()

   Key.removelistener(keyDownListener);

}
function startListeners()
{
   keyDownListener = new Object();
   keyDownListener.onKeyDown = function()
   {
      switch(Key.getCode())
      {
         case 39:
            handleKeyRight();
            break;
         case 37:
            handleKeyLeft();
            break;
         case 38:
            handleKeyUp();
            break;
         case 40:
            handleKeyDown();
            break;
         case 13:
            handleKeyEnter();
            break;
         case 8:
            handleKeyBack();
            break;
         case 36:
            break;
         case 35:
            handleKeyDelete();
      }
      if(!soeIsGame())
      {
         if(Key.getAscii() >= 97 && Key.getAscii() <= 122)
         {
            if(bSolveMode)
            {
               doSetSolveLetter(chr(Key.getAscii()).toUpperCase());
               soeTRACE("find:" + chr(Key.getAscii()).toUpperCase());
            }
            else
            {
               doFindLetters(chr(Key.getAscii()).toUpperCase());
               soeTRACE("find:" + chr(Key.getAscii()).toUpperCase());
            }
         }
         else
         {
            switch(chr(Key.getAscii()))
            {
               case "1":
                  doShowWhitePanels();
                  break;
               case "2":
                  doShowTossupLetters();
                  break;
               case "3":
                  bSolveMode = true;
                  doStartSolveMode();
                  break;
               case "4":
                  bSolveMode = true;
                  doEndSolveMode();
                  break;
               case "0":
                  doFindAllLetters();
               default:
                  return;
            }
         }
      }
   };
   Key.addListener(keyDownListener);
   keyUpListener = new Object();
   keyUpListener.onKeyUp = function()
   {
      switch(Key.getCode())
      {
         case 39:
         case 37:
         case 38:
         case 40:
         case 13:
         case 8:
         case 36:
         case 35:
         default:
            return;
      }
   };
   Key.addListener(keyUpListener);
}
