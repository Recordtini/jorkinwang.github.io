            break;

            break;

            break;

            break;

            handleKeyDelete();

      {
         switch(chr(Key.getAscii()))
         {

               if(bShowLetterboard)
               {

                  doHideLetterboard();
               }
               else
               {
                  bShowLetterboard = 1;
                  doShowLetterboard();
               }
               break;
            case "2":
               if(bSetVowelsActive)
               {
                  bSetVowelsActive = 0;
                  setVowelsInactive();
               }
               else
               {
                  bSetVowelsActive = 1;
                  setVowelsActive();
               }
               break;
            case "3":
               if(bSetConsonantsActive)
               {
                  bSetConsonantsActive = 0;
                  setConsonantsInactive();
               }
               else
               {
                  bSetConsonantsActive = 1;
                  setConsonantsActive();
               }
               break;
            case "4":
               if(bSetAllLettersActive)
               {
                  bSetAllLettersActive = 0;
                  setAllLettersInactive();
               }
               else
               {
                  bSetAllLettersActive = 1;
                  setAllLettersActive();
               }
               break;
            case "5":
            case "6":
            case "7":
            case "8":
            default:
               return;
         }
      }
   };
   Key.addListener(keyDownListener);
   keyUpListener = new Object();
   keyUpListener.onKeyUp = function()
   {
