function startListeners()
{
   keyDownListener = new Object();
   keyDownListener.onKeyDown = function()
   {
      tileboardTRACE("onKeyDown");
      if(oBoardControl)
      {
         switch(Key.getCode())
         {
            case 39:
               handleKeyRight();
               break;
            case 37:
               handleKeyLeft();
               break;
            case 38:
               handleKeyUp();
               break;
            case 40:
               handleKeyDown();
               break;
            case 32:
               handleKeySpace();
               break;
            case 8:
            case 36:
            case 35:
            default:
               return;
         }
      }
   };
   Key.addListener(keyDownListener);
   keyUpListener = new Object();
   keyUpListener.onKeyUp = function()
   {
      switch(Key.getCode())
      {
         case 39:
         case 37:
         case 38:
         case 40:
         case 32:
         case 8:
         case 36:
         case 35:
         default:
            return;
      }
   };
   Key.addListener(keyUpListener);
}
